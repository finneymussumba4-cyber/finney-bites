class User(Base):
    __tablename__ = "users"

    id: Mapped[int] = mapped_column(primary_key=True)
    full_name: Mapped[str] = mapped_column(String(150))
    email: Mapped[str] = mapped_column(
        String(255), unique=True, index=True
    )
    phone: Mapped[str] = mapped_column(
        String(20), unique=True
    )
    password_hash: Mapped[str] = mapped_column(String(255))
    role: Mapped[str] = mapped_column(
        String(20), default="customer"
    )
    is_active: Mapped[bool] = mapped_column(
        Boolean, default=True
    )
    created_at: Mapped[datetime] = mapped_column(
        DateTime, server_default=func.now()
    )


class Category(Base):
    __tablename__ = "categories"

    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str] = mapped_column(
        String(100), unique=True
    )
    description: Mapped[str | None] = mapped_column(Text)
    image: Mapped[str | None] = mapped_column(String(255))
    is_active: Mapped[bool] = mapped_column(
        Boolean, default=True
    )

    products = relationship("Product", back_populates="category")


class Product(Base):

    is_featured: Mapped[bool] = mapped_column(
        Boolean, default=False
    )
    created_at: Mapped[datetime] = mapped_column(

def normalize_phone(phone: str) -> str:
    digits = "".join(character for character in phone if character.isdigit())

    if digits.startswith("0"):
        return "254" + digits[1:]

    if digits.startswith("254"):
        return digits

    return digits


@router.post("/{order_number}/payment-reference")
def submit_payment_reference(
    order_number: str,
    data: PaymentReferenceSubmit,
    db: Session = Depends(get_db)
):
    order = db.query(Order).filter(
        Order.order_number == order_number
    ).first()

    if not order:
        raise HTTPException(status_code=404, detail="Order not found")

    if normalize_phone(data.customer_phone) != normalize_phone(order.customer_phone):
        raise HTTPException(status_code=403, detail="Phone number does not match this order")

    if order.status == "cancelled":
        raise HTTPException(status_code=400, detail="Cancelled orders cannot receive payments")

    if order.payment_status == "paid":
        raise HTTPException(status_code=400, detail="This order is already paid")

    reference = data.mpesa_reference.strip().upper()

    existing = db.query(Order).filter(
        Order.mpesa_reference == reference,
        Order.id != order.id
    ).first()

    if existing:
        raise HTTPException(status_code=409, detail="This M-Pesa reference has already been submitted")

    order.mpesa_reference = reference
    order.payment_status = "unverified"

    try:
        db.commit()
    except IntegrityError:
        db.rollback()
        raise HTTPException(status_code=409, detail="This M-Pesa reference has already been used")

    return {
        "message": "Payment reference submitted successfully. Awaiting administrator verification.",
        "order_number": order.order_number,
        "payment_status": order.payment_status
    }

@router.post("/api/reviews/{order_number}")
def submit_review(
    order_number: str,
    data: ReviewCreate,
    db: Session = Depends(get_db)
):
    order = db.query(Order).filter(
        Order.order_number == order_number
    ).first()

    if not order:
        raise HTTPException(status_code=404, detail="Order not found")

    if normalize_phone(data.customer_phone) != normalize_phone(order.customer_phone):
        raise HTTPException(status_code=403, detail="Phone number does not match this order")

    if order.status != "completed" or order.payment_status != "paid":
        raise HTTPException(
            status_code=400,
            detail="Only paid and completed orders can be reviewed"
        )

    existing = db.query(Review).filter(
        Review.order_id == order.id
    ).first()

    if existing:
        raise HTTPException(status_code=409, detail="This order already has a review")

    review = Review(
        order_id=order.id,
        rating=data.rating,
        comment=data.comment.strip(),
        is_approved=False
    )

    try:
        db.add(review)
        db.commit()
        return {"message": "Review submitted and awaiting approval"}
    except IntegrityError:
        db.rollback()
        raise HTTPException(status_code=409, detail="This order already has a review")

from fastapi import HTTPException
from sqlalchemy import update
from app.models import Order, OrderItem, InventoryLog


@router.post("/orders/{order_number}/confirm")
def confirm_order(
    order_number: str,
    admin: User = Depends(get_current_admin),
    db: Session = Depends(get_db)
):
    try:
        order = db.query(Order).filter(
            Order.order_number == order_number
        ).first()

        if not order:
            raise HTTPException(status_code=404, detail="Order not found")

        if order.stock_deducted:
            return {
                "message": "Order was already confirmed. Stock was not deducted again.",
                "order_number": order.order_number
            }

        if order.status == "cancelled":
            raise HTTPException(status_code=400, detail="Cancelled orders cannot be confirmed")

        for item in order.items:
            result = db.execute(
                update(Product)
                .where(
                    Product.id == item.product_id,
                    Product.stock_quantity >= item.quantity
                )
                .values(
                    stock_quantity=Product.stock_quantity - item.quantity
                )
            )

            if result.rowcount != 1:
                raise HTTPException(
                    status_code=400,
                    detail=f"Insufficient stock for {item.product_name}"
                )

            db.add(InventoryLog(
                product_id=item.product_id,
                change_quantity=-item.quantity,
                reason=f"Order {order.order_number} confirmed"
            ))

        order.stock_deducted = True
        order.status = "confirmed"

        db.commit()

        return {
            "message": "Order confirmed and stock deducted successfully",
            "order_number": order.order_number,
            "status": order.status,
            "stock_deducted": order.stock_deducted
        }

    except HTTPException:
        db.rollback()
        raise
    except Exception:
        db.rollback()
        raise HTTPException(
            status_code=500,
            detail="Unable to confirm order"
        )

from app.schemas.payment import PaymentVerification


@router.post("/orders/{order_number}/verify-payment")
def verify_order_payment(
    order_number: str,
    data: PaymentVerification,
    admin: User = Depends(get_current_admin),
    db: Session = Depends(get_db)
):
    order = db.query(Order).filter(
        Order.order_number == order_number
    ).first()

    if not order:
        raise HTTPException(status_code=404, detail="Order not found")

    if order.status == "cancelled":
        raise HTTPException(status_code=400, detail="Cancelled orders cannot be paid")

    if order.payment_status == "paid":
        raise HTTPException(status_code=400, detail="Payment is already verified")

    if not order.mpesa_reference:
        raise HTTPException(status_code=400, detail="Customer has not submitted an M-Pesa reference")

    if data.mpesa_reference.strip().upper() != order.mpesa_reference:
        raise HTTPException(status_code=400, detail="M-Pesa reference does not match")

    if not data.confirmed_in_statement:
        raise HTTPException(status_code=400, detail="Confirm the transaction in the M-Pesa statement first")

    received_amount = Decimal(str(data.received_amount)).quantize(Decimal("0.01"))

    if received_amount != order.total_amount:
        raise HTTPException(status_code=400, detail="Received amount does not match the order total")

    order.payment_status = "paid"

    try:
        db.commit()
    except Exception:
        db.rollback()
        raise HTTPException(status_code=500, detail="Unable to verify payment")

    return {
        "message": "Payment successfully verified",
        "order_number": order.order_number,
        "payment_status": order.payment_status,
        "verified_by": admin.full_name
    }

from app.schemas.product import ProductUpdate


@router.patch("/products/{product_id}")
def update_product(
    product_id: int,
    data: ProductUpdate,
    admin: User = Depends(get_current_admin),
    db: Session = Depends(get_db)
):
    product = db.query(Product).filter(
        Product.id == product_id
    ).first()

    if not product:
        raise HTTPException(status_code=404, detail="Product not found")

    updates = data.model_dump(exclude_unset=True)

    if not updates:
        raise HTTPException(status_code=400, detail="No updates provided")

    if "category_id" in updates:
        category = db.query(Category).filter(
            Category.id == updates["category_id"],
            Category.is_active == True
        ).first()

        if not category:
            raise HTTPException(
                status_code=400,
                detail="Invalid or inactive category"
            )

    try:
        if "stock_quantity" in updates:
            difference = updates["stock_quantity"] - product.stock_quantity

            if difference != 0:
                db.add(InventoryLog(
                    product_id=product.id,
                    change_quantity=difference,
                    reason="Admin stock adjustment"
                ))

        for field, value in updates.items():
            setattr(product, field, value)

        db.commit()
        db.refresh(product)

        return {
            "message": "Product updated successfully",
            "product_id": product.id,
            "name": product.name,
            "price": str(product.price),
            "stock_quantity": product.stock_quantity
        }

    except Exception:
        db.rollback()
        raise HTTPException(
            status_code=500,
            detail="Unable to update product"
        )

    async function editProduct(productId, currentPrice, currentStock) {
        const newPrice = prompt("Enter the new product price (KSh):", currentPrice);

        if (newPrice === null) return;

        const newStock = prompt("Enter the new stock quantity:", currentStock);

        if (newStock === null) return;

        const price = Number(newPrice);
        const stock = Number(newStock);

        if (!Number.isFinite(price) || price <= 0) {
            alert("Enter a valid price greater than zero.");
            return;
        }

        if (!Number.isInteger(stock) || stock < 0) {
            alert("Enter a valid stock quantity (0 or more).");
            return;
        }

        try {
            const response = await fetch(
                "/api/admin/products/" + productId,
                {
                    method: "PATCH",
                    headers: {
                        "Content-Type": "application/json"
                    },
                    credentials: "same-origin",
                    body: JSON.stringify({
                        price: price,
                        stock_quantity: stock
                    })
                }
            );

            const result = await response.json();

            if (!response.ok) {
                throw new Error(result.detail || "Product update failed.");
            }

            alert("Product price and quantity updated successfully!");
            await loadProducts();

            alert(error.message);

                    order.order_number,
                    order.customer_name + " (" + order.customer_phone + ")",
                    items,
                    "KSh " + Number(order.total_amount).toLocaleString(),
                    order.mpesa_reference || "Not submitted",
                    order.payment_status,
                    order.status
                ];

                values.forEach(value => {
                    const cell = document.createElement("td");
                    cell.style.padding = "12px";
                    cell.textContent = value;
                    row.appendChild(cell);
                });

                const actions = document.createElement("td");
                actions.style.padding = "12px";
                if (order.status === "received") {
                    const confirmButton = document.createElement("button");
                    confirmButton.textContent = "Confirm Order";
                    confirmButton.onclick = () => confirmOrder(order.order_number);
                    actions.appendChild(confirmButton);
                }

                if (order.payment_status !== "paid" &&
                    order.mpesa_reference &&
                    order.status !== "cancelled") {
                    const verifyButton = document.createElement("button");
                    verifyButton.textContent = "Verify Payment";
                    verifyButton.style.marginTop = "6px";
                    verifyButton.onclick = () => verifyPayment(
                        order.order_number,
                        order.mpesa_reference,
                        order.total_amount
                    );
                    actions.appendChild(verifyButton);
                }

                if (order.status === "confirmed" &&
                    order.payment_status === "paid") {
                    const completeButton = document.createElement("button");
                    completeButton.textContent = "Complete Order";
                    completeButton.style.marginTop = "6px";
                    completeButton.onclick = () => completeOrder(order.order_number);
                    actions.appendChild(completeButton);
                }

                if (actions.children.length === 0) {
                    actions.textContent = "No available actions.";
                }

                row.appendChild(actions);
                table.appendChild(row);
            });

        } catch (error) {
            message.textContent = error.message;
        }
    }

    async function confirmOrder(orderNumber) {
        if (!confirm("Confirm order " + orderNumber + "? Stock will be deducted.")) {
            return;
        }

        try {
            const response = await fetch(
                "/api/admin/orders/" + encodeURIComponent(orderNumber) + "/confirm",
                {
                    method: "POST",
                    credentials: "same-origin"
                }
            );

            const result = await response.json();

            if (!response.ok) {
                throw new Error(result.detail || "Order confirmation failed.");
            }

            alert(result.message);
            await loadOrders();
            await loadMessages();
            await loadProducts();

        } catch (error) {
            alert(error.message);
        }
    }

    async function verifyPayment(orderNumber, currentReference, orderTotal) {
        const reference = prompt(
            "Enter the M-Pesa reference:",
            currentReference
        );

        if (reference === null) return;

        const amount = prompt(
            "Enter the amount received (KSh):",
            orderTotal
        );

        if (amount === null) return;

        if (!confirm("Have you confirmed this transaction in your M-Pesa statement?")) {
            alert("Payment was not verified.");
            return;
        }

        const receivedAmount = Number(amount);

        if (!Number.isFinite(receivedAmount) || receivedAmount <= 0) {
            alert("Enter a valid payment amount.");
            return;
        }

        try {
            const response = await fetch(
                "/api/admin/orders/" + encodeURIComponent(orderNumber) + "/verify-payment",
                {
                    method: "POST",
                    headers: {"Content-Type": "application/json"},
                    credentials: "same-origin",
                    body: JSON.stringify({
                        mpesa_reference: reference.trim().toUpperCase(),
                        received_amount: receivedAmount,
                        confirmed_in_statement: true
                    })
                }
            );

            const result = await response.json();

            if (!response.ok) {
                throw new Error(result.detail || "Payment verification failed.");
            }

            alert(result.message);
            await loadOrders();
            await loadMessages();

        } catch (error) {
            alert(error.message);
        }
    }
    async function completeOrder(orderNumber) {
        if (!confirm("Mark order " + orderNumber + " as completed?")) return;

        try {
            const response = await fetch(
                "/api/admin/orders/" + encodeURIComponent(orderNumber) + "/complete",
                {
                    method: "POST",
                    credentials: "same-origin"
                }
            );

            const result = await response.json();

            if (!response.ok) {
                throw new Error(result.detail || "Unable to complete order.");
            }

            alert(result.message);
            await loadOrders();
            await loadMessages();

        } catch (error) {
            alert(error.message);
        }
    }
    async function loadMessages() {
        const table = document.getElementById("messagesTable");
        const message = document.getElementById("messagesMessage");

        table.innerHTML = "";
        message.textContent = "Loading customer messages...";

        try {
            const response = await fetch("/api/admin/messages", {
                credentials: "same-origin"
            });

            const messages = await response.json();

            if (!response.ok) {
                throw new Error(messages.detail || "Unable to load messages.");
            }

            if (messages.length === 0) {
                message.textContent = "No customer messages found.";
                table.innerHTML =
                    '<tr><td colspan="7">No messages available.</td></tr>';
                return;
            }

            message.textContent = messages.length + " messages loaded.";

            messages.forEach(item => {
                const row = document.createElement("tr");

                [
                    item.name,
                    item.email,
                    item.phone || "Not provided",
                    item.subject,
                    item.message,
                    item.status
                ].forEach(value => {
                    const cell = document.createElement("td");
                    cell.style.padding = "12px";
                    cell.textContent = value;
                    row.appendChild(cell);
                });

                const actions = document.createElement("td");
                actions.style.padding = "12px";

                if (item.status === "new") {
                    const readButton = document.createElement("button");
                    readButton.textContent = "Mark Read";
                    readButton.onclick = () =>
                        updateMessageStatus(item.id, "read");
                    actions.appendChild(readButton);
                }

                if (item.status !== "resolved") {
                    const resolveButton = document.createElement("button");
                    resolveButton.textContent = "Resolve";
                    resolveButton.style.marginTop = "6px";
                    resolveButton.onclick = () =>
                        updateMessageStatus(item.id, "resolved");
                    actions.appendChild(resolveButton);
                }

                if (actions.children.length === 0) {
                    actions.textContent = "Resolved";
                }

                row.appendChild(actions);
                table.appendChild(row);
            });

        } catch (error) {
            message.textContent = error.message;
        }
    }

    async function updateMessageStatus(messageId, newStatus) {
        try {
            const response = await fetch(
                "/api/admin/messages/" + messageId +
                "/status?status=" + encodeURIComponent(newStatus),
                {
                    method: "PATCH",
                    credentials: "same-origin"
                }
            );

            const result = await response.json();

            if (!response.ok) {
                throw new Error(result.detail || "Unable to update message.");
            }

            alert(result.message);
            await loadMessages();

        } catch (error) {
            alert(error.message);
        }
    }
    async function loadReviews() {
        const table = document.getElementById("reviewsTable");
        const message = document.getElementById("reviewsMessage");

        table.innerHTML = "";
        message.textContent = "Loading reviews...";

        try {
            const response = await fetch("/api/admin/reviews/pending", {
                credentials: "same-origin"
            });

            const reviews = await response.json();

            if (!response.ok) {
                throw new Error(reviews.detail || "Unable to load reviews.");
            }

            if (reviews.length === 0) {
                message.textContent = "No pending reviews found.";
                table.innerHTML =
                    '<tr><td colspan="6">No pending reviews available.</td></tr>';
                return;
            }

            message.textContent = reviews.length + " pending reviews loaded.";

            reviews.forEach(review => {
                const row = document.createElement("tr");

                const values = [
                    review.id,
                    review.order_id,
                    review.rating + " / 5",
                    review.comment,
                    review.created_at
                        ? new Date(review.created_at).toLocaleString()
                        : "Not available"
                ];

                values.forEach(value => {
                    const cell = document.createElement("td");
                    cell.style.padding = "12px";
                    cell.textContent = value;
                    row.appendChild(cell);
                });

                const actions = document.createElement("td");
                actions.style.padding = "12px";

                const approveButton = document.createElement("button");
                approveButton.textContent = "Approve";
                approveButton.onclick = () => approveReview(review.id);

                actions.appendChild(approveButton);
                row.appendChild(actions);
                table.appendChild(row);
            });

        } catch (error) {
            message.textContent = error.message;
        }
    }

    async function approveReview(reviewId) {
        if (!confirm("Approve this customer review?")) return;

        try {
            const response = await fetch(
                "/api/admin/reviews/" + reviewId + "/approve",
                {
                    method: "POST",
                    credentials: "same-origin"
                }
            );

            const result = await response.json();

            if (!response.ok) {
                throw new Error(result.detail || "Unable to approve review.");
            }

            alert(result.message);
            await loadReviews();

        } catch (error) {
            alert(error.message);
        }
    }

    async function loadWhatsAppStatus() {
        const configured = document.getElementById("whatsappConfigured");
        const mode = document.getElementById("whatsappMode");
        const message = document.getElementById("whatsappMessage");

        configured.textContent = "Checking...";
        mode.textContent = "Checking...";
        message.textContent = "";

        try {
            const response = await fetch("/api/admin/whatsapp-status", {
                credentials: "same-origin"
            });

            const result = await response.json();

            if (!response.ok) {
                throw new Error(result.detail || "Unable to check WhatsApp status.");
            }

            configured.textContent = result.configured ? "Configured" : "Not configured";
            mode.textContent = result.mode.toUpperCase();
            message.textContent = result.message;

        } catch (error) {
            configured.textContent = "Unavailable";
            mode.textContent = "Unknown";
            message.textContent = error.message;
        }
    }
